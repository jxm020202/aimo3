def can_slide(dom_idx, selection_set, dominos, n):
    d = dominos[dom_idx]
    orient = d['orient']
    r,c = d['r'], d['c']
    # occupancy set
    occ = set()
    for i in selection_set:
        for cell in dominos[i]['cells']:
            occ.add(cell)
    # check each direction
    if orient == 'H':
        # left
        if c-1 >= 0:
            if (r,c-1) not in occ:
                # both cells left are empty? Actually need both cells left? For horizontal sliding left, only need cell (r,c-1) empty (the new left cell). The rightmost cell (r,c) will be vacated, so only need left cell empty.
                # sliding left possible if (r,c-1) empty
                return True, 'left'
        # right
        if c+2 < n:
            if (r,c+2) not in occ:
                return True, 'right'
        # up
        if r-1 >= 0:
            # need both cells above empty
            if (r-1,c) not in occ and (r-1,c+1) not in occ:
                return True, 'up'
        # down
        if r+1 < n:
            if (r+1,c) not in occ and (r+1,c+1) not in occ:
                return True, 'down'
    else: # V
        # up
        if r-1 >= 0:
            if (r-1,c) not in occ:
                return True, 'up'
        # down
        if r+2 < n:
            if (r+2,c) not in occ:
                return True, 'down'
        # left
        if c-1 >= 0:
            if (r,c-1) not in occ and (r+1,c-1) not in occ:
                return True, 'left'
        # right
        if c+1 < n:
            if (r,c+1) not in occ and (r+1,c+1) not in occ:
                return True, 'right'
    return False, None
