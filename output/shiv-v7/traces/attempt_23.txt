def board_repr(placements):
    board = [['.' for _ in range(7)] for _ in range(7)]
    for idx, (typ, r, c) in enumerate(placements):
        if typ == 'H':
            board[r][c] = 'H'
            board[r][c+1] = 'h'
        else:
            board[r][c] = 'V'
            board[r+1][c] = 'v'
    return "\n".join("".join(row) for row in board)
