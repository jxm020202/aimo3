num_vars = len(placements)
A_rows = []
lb = []
ub = []

# Overlap constraints: for each cell, sum <=1
for cell in range(n*n):
    coeff = np.zeros(num_vars, dtype=int)
    for i in cell_to_placements[cell]:
        coeff[i] = 1
    A_rows.append(coeff)
    lb.append(-np.inf)  # no lower bound
    ub.append(1)

# Blocking constraints
for i, p in enumerate(placements):
    for d, block_cells in p['block'].items():
        if len(block_cells)==0:
            continue  # automatically blocked
        coeff = np.zeros(num_vars, dtype=int)
        # sum over placements covering any block cell
        for cell in block_cells:
            for j in cell_to_placements[cell]:
                coeff[j] = 1  # note may duplicate but set to 1
        # subtract x_i
        coeff[i] -= 1
        # inequality: sum - x_i >= 0
        A_rows.append(coeff)
        lb.append(0)
        ub.append(np.inf)

A = np.vstack(A_rows)
lb = np.array(lb, dtype=float)
ub = np.array(ub, dtype=float)

c = -np.ones(num_vars)  # maximize sum x_i => minimize -sum
# integrality
integrality = np.ones(num_vars, dtype=int)

# bounds for variables 0-1
bounds = Bounds(0, 1)

# Solve MILP
res = milp(c=c, constraints=LinearConstraint(A, lb, ub), bounds=bounds, integrality=integrality, options={'disp': False})
print(res.success, res.message)

def can_slide(p_idx, direction):
    # returns True if domino can slide one step in direction given other dominoes fixed
    p = placements[p_idx]
    cells = p['cells']
    # compute new cells after move
    dr, dc = 0,0
    if direction=='L':
        dr, dc = 0, -1
    elif direction=='R':
        dr, dc = 0, 1
    elif direction=='U':
        dr, dc = -1, 0
    elif direction=='D':
        dr, dc = 1, 0
    new_cells = []
    for cell in cells:
        r = cell // n
        c = cell % n
        nr, nc = r+dr, c+dc
        if not (0 <= nr < n and 0 <= nc < n):
            return False  # would go off board, cannot slide
        new_cells.append((nr,nc))
    # check that new cells are empty (not occupied by other domino)
    for nr,nc in new_cells:
        if board[nr,nc] != -1 and board[nr,nc] != board[cells[0]//n, cells[0]%n]:
            return False
    # also need that the original cells vacated are not needed? sliding allowed regardless of emptiness after move.
    return True
