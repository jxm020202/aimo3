        if c > 0:

        if c > 0:
            dirs.append([cell_index(r, c-1), cell_index(r+1, c-1)])

        if c+2 < n:

    else:
        if c+1 < n:
            dirs.append([cell_index(r, c+1), cell_index(r+1, c+1)])

def build_constraints(target_m):
    # Build constraints matrices for given target number of dominos
    A_rows = []
    lb = []
    ub = []
    # Non-overlap constraints
    for cell in range(n*n):
        coeff = np.zeros(num_vars, dtype=int)
        for i, cells in enumerate(placement_cells):
            if cell in cells:
                coeff[i] = 1
        A_rows.append(coeff)
        lb.append(-np.inf)
        ub.append(1)
    # Immobility constraints
    for i in range(num_vars):
        neighbor_dirs = neighbor_sets_for_placement(i)
        for neigh in neighbor_dirs:
            coeff = np.zeros(num_vars, dtype=int)
            coeff[i] = -1
            # sum over placements covering any neighbor cell
            for k, cells in enumerate(placement_cells):
                if any(cell in cells for cell in neigh):
                    coeff[k] += 1
            A_rows.append(coeff)
            lb.append(0)
            ub.append(np.inf)
    # Equality sum = target_m
    coeff = np.ones(num_vars, dtype=int)
    A_rows.append(coeff)
    lb.append(target_m)
    ub.append(target_m)
    A = np.vstack(A_rows)
    lb_arr = np.array(lb, dtype=float)
    ub_arr = np.array(ub, dtype=float)
    return A, lb_arr, ub_arr

def can_slide(board, typ, r, c):
    # board occupancy: True if occupied
    occ = [[False]*n for _ in range(n)]
    # fill occupancy from placements_selected
    for idx in placements_selected:
        t, rr, cc = placements[idx]
        if t == 'H':
            occ[rr][cc] = True
            occ[rr][cc+1] = True
        else:
            occ[rr][cc] = True
            occ[rr+1][cc] = True
    # check slide directions
    if typ == 'H':
        # left
        if c > 0:
            if not occ[r][c-1]:
                return True
        # right
        if c+2 < n:
            if not occ[r][c+2]:
                return True
        # up
        if r > 0:
            if not occ[r-1][c] and not occ[r-1][c+1]:
                return True
        # down
        if r+1 < n:
            if not occ[r+1][c] and not occ[r+1][c+1]:
                return True
    else:
        # vertical
        # up
        if r > 0:
            if not occ[r-1][c]:
                return True
        # down
        if r+2 < n:
            if not occ[r+2][c]:
                return True
        # left
        if c > 0:
            if not occ[r][c-1] and not occ[r+1][c-1]:
                return True
        # right
        if c+1 < n:
            if not occ[r][c+1] and not occ[r+1][c+1]:
                return True
    return False
