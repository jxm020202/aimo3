def build_constraints():
    num_vars = len(placements)
    rows = []
    # cell occupancy <=1
    for r in range(7):
        for c in range(7):
            coeff = np.zeros(num_vars, dtype=int)
            for idx in cell_to_placements.get((r,c), []):
                coeff[idx] = 1
            rows.append((coeff, -np.inf, 1))
    # neighbor group constraints
    for i, p in enumerate(placements):
        typ, r, c = p
        groups = []
        if typ == 'H':
            # left
            if c-1 >= 0:
                groups.append([(r, c-1)])
            # right
            if c+2 < 7:
                groups.append([(r, c+2)])
            # up
            if r-1 >= 0:
                groups.append([(r-1, c), (r-1, c+1)])
            # down
            if r+1 < 7:
                groups.append([(r+1, c), (r+1, c+1)])
        else:  # V
            # up
            if r-1 >= 0:
                groups.append([(r-1, c)])
            # down

                groups.append([(r+2, c)])
            # left
            if c-1 >= 0:
                groups.append([(r, c-1), (r+1, c-1)])
            # right
            if c+1 < 7:
                groups.append([(r, c+1), (r+1, c+1)])
        # for each group, add constraint sum occupancy >= x_i
        for group in groups:

            rows.append((coeff, 0, np.inf))
