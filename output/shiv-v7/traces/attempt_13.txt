def add_blocking_constraint(idx, blocking_cells):
    # idx: placement index of domino that must be blocked
    # blocking_cells: list of cell coordinates that need to be occupied (at least one)
    # Build constraint: x_idx - sum_{vars covering any blocking cell} <= 0
    coeff = np.zeros(n)
    coeff[idx] = 1.0
    # For each blocking cell, subtract all vars covering it
    for cell in blocking_cells:
        for var in cell_to_vars.get(cell, []):
            coeff[var] -= 1.0
    constraints.append((coeff, 0.0))

def can_slide(idx, board):
    # board: 7x7 with -1 empty else placement index
    # returns True if domino can slide in any direction
    # Find its cells
    cells = placements[idx]['cells']
    typ = placements[idx]['type']
    r,c = placements[idx]['r'], placements[idx]['c']
    # Directions: up, down, left, right
    # For each direction, check if target cells are empty (and within board)
    # For horizontal domino:
    if typ=='H':
        # left: need cell (r,c-1) empty and within board
        if c-1 >=0 and board[r,c-1]==-1:
            return True
        # right: need (r,c+2)
        if c+2 <7 and board[r,c+2]==-1:
            return True
        # up: need both (r-1,c) and (r-1,c+1) empty
        if r-1 >=0 and board[r-1,c]==-1 and board[r-1,c+1]==-1:
            return True
        # down: need both (r+1,c) and (r+1,c+1) empty
        if r+1 <7 and board[r+1,c]==-1 and board[r+1,c+1]==-1:
            return True
    else: # V
        # up: need (r-1,c)
        if r-1 >=0 and board[r-1,c]==-1:
            return True

        if r+2 <7 and board[r+2,c]==-1:
            return True
        # left: need both (r,c-1) and (r+1,c-1) empty
        if c-1 >=0 and board[r,c-1]==-1 and board[r+1,c-1]==-1:
            return True

        if c+1 <7 and board[r,c+1]==-1 and board[r+1,c+1]==-1:

    return False
