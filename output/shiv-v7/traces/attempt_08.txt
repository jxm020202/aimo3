def is_blocked(dom_idx, x_vec):
    # returns True if domino cannot slide in any direction given other dominos (x_vec)
    c1,c2 = dominos[dom_idx]
    r1,c1c = c1
    r2,c2c = c2
    # orientation
    if r1 == r2:  # horizontal
        r = r1
        left_c = min(c1c, c2c)
        # left
        if left_c - 1 >= 0:
            cell = (r, left_c-1)
            # if empty -> can slide
            if sum(x_vec[j] for j in cell_to_dominos[cell]) == 0:
                return False
        # right
        if left_c + 2 <= 6:
            cell = (r, left_c+2)
            if sum(x_vec[j] for j in cell_to_dominos[cell]) == 0:
                return False
        # up
        if r - 1 >= 0:
            cells_up = [(r-1, left_c), (r-1, left_c+1)]
            # if both empty -> can slide up
            if all(sum(x_vec[j] for j in cell_to_dominos[cell]) == 0 for cell in cells_up):
                return False
        # down
        if r + 1 <= 6:
            cells_down = [(r+1, left_c), (r+1, left_c+1)]
            if all(sum(x_vec[j] for j in cell_to_dominos[cell]) == 0 for cell in cells_down):
                return False
    else:  # vertical
        c = c1c
        top_r = min(r1, r2)
        # up
        if top_r - 1 >= 0:
            cell = (top_r-1, c)
            if sum(x_vec[j] for j in cell_to_dominos[cell]) == 0:
                return False
        # down
        if top_r + 2 <= 6:
            cell = (top_r+2, c)
            if sum(x_vec[j] for j in cell_to_dominos[cell]) == 0:
                return False
        # left
        if c - 1 >= 0:
            cells_left = [(top_r, c-1), (top_r+1, c-1)]
            if all(sum(x_vec[j] for j in cell_to_dominos[cell]) == 0 for cell in cells_left):
                return False
        # right
        if c + 1 <= 6:
            cells_right = [(top_r, c+1), (top_r+1, c+1)]
            if all(sum(x_vec[j] for j in cell_to_dominos[cell]) == 0 for cell in cells_right):
                return False
    return True
