def build_ilp():
    rows = cols = 7
    h_vars = [(i,j) for i in range(rows) for j in range(cols-1)]
    v_vars = [(i,j) for i in range(rows-1) for j in range(cols)]
    var_list = [('h',i,j) for (i,j) in h_vars] + [('v',i,j) for (i,j) in v_vars]
    n = len(var_list)
    # cell covering mapping
    cell_cover = collections.defaultdict(list)
    for idx, (typ,i,j) in enumerate(var_list):
        if typ == 'h':
            cells = [(i,j),(i,j+1)]
        else:
            cells = [(i,j),(i+1,j)]
        for cell in cells:
            cell_cover[cell].append(idx)
    constraints = []
    lb = []
    ub = []
    # non-overlap per cell
    for i in range(rows):
        for j in range(cols):
            vars_cov = cell_cover[(i,j)]
            row = np.zeros(n, dtype=int)
            row[vars_cov] = 1
            constraints.append(row)
            lb.append(-np.inf)
            ub.append(1)
    # blocking constraints
    def add_pair_block(var_idx, cells):
        # cells is list of cells where at least one must be occupied
        # if empty list (offboard) ignore
        # constraint: sum covering of these cells >= var_idx
        row = np.zeros(n, dtype=int)
        row[var_idx] = -1
        for cell in cells:
            i,j = cell
            if 0 <= i < rows and 0 <= j < cols:
                for cov in cell_cover[(i,j)]:
                    row[cov] += 1
        # If none of the cells are on board (i.e., all offboard), then constraint is trivially satisfied, skip.
        # But we need to ensure that at least one cell is on board; else no constraint.
        # We'll add constraint only if there is at least one on-board cell.
        if any(0 <= cell[0] < rows and 0 <= cell[1] < cols for cell in cells):
            constraints.append(row)
            lb.append(0)
            ub.append(np.inf)
    # iterate dominos
    for idx, (typ,i,j) in enumerate(var_list):
        if typ == 'h':
            # left block: cell (i,j-1) if on board
            if j-1 >= 0:
                add_pair_block(idx, [(i,j-1)])
            # right block: cell (i,j+2) if on board
            if j+2 < cols:
                add_pair_block(idx, [(i,j+2)])
            # up block: cells (i-1,j) and (i-1,j+1) if on board
            if i-1 >= 0:
                add_pair_block(idx, [(i-1,j), (i-1,j+1)])
            # down block: cells (i+1,j) and (i+1,j+1) if on board
            if i+1 < rows:
                add_pair_block(idx, [(i+1,j), (i+1,j+1)])
        else:  # vertical
            # up block: cell (i-1,j)
            if i-1 >= 0:
                add_pair_block(idx, [(i-1,j)])
            # down block: cell (i+2,j)
            if i+2 < rows:
                add_pair_block(idx, [(i+2,j)])
            # left block: cells (i,j-1) and (i+1,j-1)
            if j-1 >= 0:
                add_pair_block(idx, [(i,j-1), (i+1,j-1)])
            # right block: cells (i,j+1) and (i+1,j+1)
            if j+1 < cols:
                add_pair_block(idx, [(i,j+1), (i+1,j+1)])
    A = np.vstack(constraints).astype(float)
