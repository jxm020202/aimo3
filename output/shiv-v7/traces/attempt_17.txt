def check_solution(n, placements_selected):
    # create board occupancy
    occ = [[None]*n for _ in range(n)]
    for idx, (typ, r, c) in enumerate(placements_selected):
        if typ == 'H':
            occ[r][c] = occ[r][c+1] = idx
        else:
            occ[r][c] = occ[r+1][c] = idx
    # check non-overlap already satisfied
    # check each domino blocked
    for (typ, r, c) in placements_selected:
        if typ == 'H':
            # left
            if c > 0:
                if occ[r][c-1] is None:
                    return False
            # right
            if c+2 < n:
                if occ[r][c+2] is None:
                    return False
            # up
            if r > 0:
                if occ[r-1][c] is None or occ[r-1][c+1] is None:
                    return False
            # down
            if r+2 < n:
                if occ[r+2][c] is None or occ[r+2][c+1] is None:
                    return False
        else: # V
            if c > 0:
                if occ[r][c-1] is None or occ[r+1][c-1] is None:
                    return False
            if c+1 < n:
                if occ[r][c+1] is None or occ[r+1][c+1] is None:
                    return False
            if r > 0:
                if occ[r-1][c] is None:
                    return False
            if r+2 < n:
                if occ[r+2][c] is None:
                    return False
    return True

def construct_pattern():

    empty = {(0,0),(0,6),(6,0),(6,6),(3,3)}
    occ = [[None]*n for _ in range(n)]
    placements = []
    # simple greedy fill remaining squares with dominos (horizontal first)
    for r in range(n):
        c=0
        while c < n-1:
            if (r,c) not in empty and (r,c+1) not in empty and occ[r][c] is None and occ[r][c+1] is None:
                placements.append(('H', r, c))
                occ[r][c]=occ[r][c+1]=len(placements)-1
                c+=2
            else:
                c+=1
    # fill vertical for remaining cells
    for c in range(n):
        r=0
        while r < n-1:
            if (r,c) not in empty and (r+1,c) not in empty and occ[r][c] is None and occ[r+1][c] is None:
                placements.append(('V', r, c))
                occ[r][c]=occ[r+1][c]=len(placements)-1
                r+=2
            else:
                r+=1
    return placements, empty

def extend_to_target(initial_sel, target_m):
    n=7
    # board occupancy
    occ = [[None]*n for _ in range(n)]
    for idx, (typ,r,c) in enumerate(initial_sel):
        if typ=='H':
            occ[r][c]=occ[r][c+1]=idx
        else:
            occ[r][c]=occ[r+1][c]=idx
    # generate all possible placements not overlapping
    all_places = []
    for r in range(n):
        for c in range(n-1):
            all_places.append(('H', r, c))
    for r in range(n-1):
        for c in range(n):
            all_places.append(('V', r, c))
    # filter out those overlapping with current
    used=set()
    for i, (typ,r,c) in enumerate(initial_sel):
        used.add(i)
    # we need to add more placements not overlapping
    # simple search: try adding any placement that doesn't overlap
    added=[]
    for place in all_places:
        if len(initial_sel)+len(added) >= target_m:
            break
        typ,r,c=place
        cells = [(r,c),(r,c+1)] if typ=='H' else [(r,c),(r+1,c)]
        if any(occ[x][y] is not None for x,y in cells):
            continue
        # add
        added.append(place)
        for x,y in cells:
            occ[x][y]=len(initial_sel)+len(added)-1
    return initial_sel+added
