    n = len(placements)

            total = sum(x[i] for i in cell_to_dominos[(r,c)])

def locked_config(board_rows, board_cols, max_dominos=5):
    # generate placements as before
    placements = []
    cell_to_dominos = collections.defaultdict(list)
    for r in range(board_rows):
        for c in range(board_cols-1):
            idx = len(placements)
            placements.append(('H', r, c))
            cell_to_dominos[(r,c)].append(idx)
            cell_to_dominos[(r,c+1)].append(idx)
    for r in range(board_rows-1):
        for c in range(board_cols):
            idx = len(placements)
            placements.append(('V', r, c))
            cell_to_dominos[(r,c)].append(idx)
            cell_to_dominos[(r+1,c)].append(idx)
    n = len(placements)
    # iterate subsets up to max_dominos
    for k in range(1, max_dominos+1):
        for combo in itertools.combinations(range(n), k):
            # check non-overlap
            occupied = {}
            ok = True
            for i in combo:
                orient, r, c = placements[i]
                cells = [(r,c),(r,c+1)] if orient=='H' else [(r,c),(r+1,c)]
                for cell in cells:
                    if cell in occupied:
                        ok=False; break
                    occupied[cell]=i
                if not ok: break
            if not ok:
                continue
            # check blocking for each domino
            # compute occupancy set
            occ_set = set(occupied.keys())
            for i in combo:
                orient, r, c = placements[i]
                # left/right/up/down checks
                if orient=='H':
                    # left
                    if c-1>=0 and (r,c-1) not in occ_set:
                        ok=False; break
                    # right
                    if c+2<board_cols and (r,c+2) not in occ_set:
                        ok=False; break
                    # up
                    if r-1>=0:
                        if (r-1,c) not in occ_set or (r-1,c+1) not in occ_set:
                            ok=False; break
                    # down
                    if r+1<board_rows:
                        if (r+1,c) not in occ_set or (r+1,c+1) not in occ_set:
                            ok=False; break
                else:
                    # V
                    # up
                    if r-1>=0 and (r-1,c) not in occ_set:
                        ok=False; break
                    # down
                    if r+2<board_rows and (r+2,c) not in occ_set:
                        ok=False; break
                    # left
                    if c-1>=0:
                        if (r,c-1) not in occ_set or (r+1,c-1) not in occ_set:

                    # right
                    if c+1<board_cols:
                        if (r,c+1) not in occ_set or (r+1,c+1) not in occ_set:
                            ok=False; break
            if ok:
                return combo, placements
    return None, placements
