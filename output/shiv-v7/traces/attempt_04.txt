def blocking_cells(p, direction):
    typ, r, c = p
    # direction: 'L','R','U','D'
    if typ == 'H':
        if direction == 'L':
            # left of left cell
            if c-1 < 0:
                return None
            return [(r, c-1)]
        if direction == 'R':
            if c+2 > 6:
                return None
            return [(r, c+2)]
        if direction == 'U':
            if r-1 < 0:
                return None
            return [(r-1, c), (r-1, c+1)]
        if direction == 'D':
            if r+1 > 6:
                return None
            return [(r+1, c), (r+1, c+1)]
    else: # vertical
        if direction == 'U':
            if r-1 < 0:
                return None
            return [(r-1, c)]
        if direction == 'D':
            if r+2 > 6:
                return None
            return [(r+2, c)]
        if direction == 'L':
            if c-1 < 0:
                return None
            return [(r, c-1), (r+1, c-1)]
        if direction == 'R':
            if c+1 > 6:
                return None
            return [(r, c+1), (r+1, c+1)]
    return None

def can_slide(p, direction, occupied_set):
    typ, r, c = p
    if typ == 'H':
        if direction == 'L':
            if c-1 < 0:
                return False
            new_cell = (r, c-1)
            # need new_cell empty
            return new_cell not in occupied_set
        if direction == 'R':
            if c+2 > 6:
                return False
            new_cell = (r, c+2)
            return new_cell not in occupied_set
        if direction == 'U':
            if r-1 < 0:
                return False
            new_cells = [(r-1, c), (r-1, c+1)]
            return all(cell not in occupied_set for cell in new_cells)
        if direction == 'D':
            if r+1 > 6:
                return False
            new_cells = [(r+1, c), (r+1, c+1)]
            return all(cell not in occupied_set for cell in new_cells)
    else: # V
        if direction == 'U':
            if r-1 < 0:

            return new_cell not in occupied_set

    return False
