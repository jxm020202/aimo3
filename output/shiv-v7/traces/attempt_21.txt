def check_configuration(selected):
    # selected: list of placement indices
    occupied = set()
    for idx in selected:
        typ, r, c = placements[idx]
        if typ == 'H':
            cells = [(r,c), (r,c+1)]
        else:
            cells = [(r,c), (r+1,c)]
        for cell in cells:
            if cell in occupied:
                return False, f"Overlap at {cell}"
            occupied.add(cell)
    # now check each domino blocked
    for idx in selected:
        typ, r, c = placements[idx]
        # check each direction
        if typ == 'H':
            # left
            if c-1 >= 0:
                if (r, c-1) not in occupied:
                    return False, f"Domino {idx} can slide left"
            # right
            if c+2 < n:
                if (r, c+2) not in occupied:
                    return False, f"Domino {idx} can slide right"
            # up
            if r-1 >= 0:
                if (r-1, c) not in occupied and (r-1, c+1) not in occupied:
                    return False, f"Domino {idx} can slide up"
            # down
            if r+1 < n:
                if (r+1, c) not in occupied and (r+1, c+1) not in occupied:
                    return False, f"Domino {idx} can slide down"
        else: # V
            # up
            if r-1 >= 0:
                if (r-1, c) not in occupied:
                    return False, f"Domino {idx} can slide up"
            # down
            if r+2 < n:
                if (r+2, c) not in occupied:
                    return False, f"Domino {idx} can slide down"
            # left
            if c-1 >= 0:
                if (r, c-1) not in occupied and (r+1, c-1) not in occupied:
                    return False, f"Domino {idx} can slide left"
            # right
            if c+1 < n:
                if (r, c+1) not in occupied and (r+1, c+1) not in occupied:

    return True, "OK"
