    (r1,c1), (r2,c2) = cells

    if r1 == r2:  # horizontal
        r = r1
        c_left = min(c1,c2)  # leftmost column
        # left
        left_cell = (r, c_left-1)
        # right
        right_cell = (r, c_left+2)
        # up cells
        up_cells = [(r-1, c_left), (r-1, c_left+1)]
        # down cells
        down_cells = [(r+1, c_left), (r+1, c_left+1)]
        dirs = {}

            dirs['right'] = [cell_index(*right_cell)]
        else:
            dirs['right'] = None
