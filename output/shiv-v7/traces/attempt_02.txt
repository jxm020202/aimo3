# board size
n = 7

# enumerate domino placements
h_vars = []  # list of (r,c) for horizontal domino covering (r,c) and (r,c+1)
v_vars = []  # list of (r,c) for vertical domino covering (r,c) and (r+1,c)

for r in range(n):
    for c in range(n-1):
        h_vars.append((r,c))
for r in range(n-1):
    for c in range(n):
        v_vars.append((r,c))

num_h = len(h_vars)
num_v = len(v_vars)
num_vars = num_h + num_v

# mapping from cell to list of variable indices covering it
cell_to_vars = {}
def add_cell(i,j,var_idx):
    cell_to_vars.setdefault((i,j), []).append(var_idx)

for idx, (r,c) in enumerate(h_vars):
    add_cell(r,c, idx)
    add_cell(r,c+1, idx)
for idx, (r,c) in enumerate(v_vars, start=num_h):
    add_cell(r,c, idx)
    add_cell(r+1,c, idx)

A = []
lb = []
ub = []

    row = np.zeros(num_vars, dtype=int)
    for v in varlist:
        row[v] = 1
    A.append(row)
    lb.append(-np.inf)  # no lower bound
    ub.append(1)

# 2. Blocking constraints
# For each domino variable, enforce blocking conditions
def cell_occupied_sum(i,j):
    # returns row vector sum of variables covering cell (i,j)
    row = np.zeros(num_vars, dtype=int)
    for v in cell_to_vars.get((i,j), []):
        row[v] = 1
    return row

def can_domino_slide(domino, occupied_set):

    (r1,c1), (r2,c2) = cells
    if orient == 'H':
        r = r1

            if (r, c-1) not in occupied_set:
                return True

                return True
        # up
        if r-1 >= 0:
            if (r-1, c) not in occupied_set and (r-1, c+1) not in occupied_set:
                return True
        # down
        if r+1 < 7:
            if (r+1, c) not in occupied_set and (r+1, c+1) not in occupied_set:
                return True
    else:  # V

    return False

def max_dominos(n):
    # build ILP for n x n board
    h_vars=[(r,c) for r in range(n) for c in range(n-1)]
    v_vars=[(r,c) for r in range(n-1) for c in range(n)]
    num_h=len(h_vars); num_v=len(v_vars); num_vars=num_h+num_v
    cell_to_vars={}
    def add_cell(i,j,var_idx):
        cell_to_vars.setdefault((i,j), []).append(var_idx)
    for idx,(r,c) in enumerate(h_vars):
        add_cell(r,c,idx); add_cell(r,c+1,idx)
    for idx,(r,c) in enumerate(v_vars, start=num_h):
        add_cell(r,c,idx); add_cell(r+1,c,idx)
    rows=[]; lbs=[]; ubs=[]
    BIG=1000
    # non-overlap
    for (i,j), varlist in cell_to_vars.items():
        row=np.zeros(num_vars)
        for v in varlist:
            row[v]=1
        rows.append(row); lbs.append(-BIG); ubs.append(1)
    def cell_row(i,j):
        row=np.zeros(num_vars)
        for v in cell_to_vars.get((i,j), []):
            row[v]=1
        return row
    # blocking constraints
    for idx,(r,c) in enumerate(h_vars):
        if c>0:
            row=cell_row(r,c-1); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if c+2 < n:
            row=cell_row(r,c+2); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if r>0:
            row=cell_row(r-1,c)+cell_row(r-1,c+1); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if r+1 < n:
            row=cell_row(r+1,c)+cell_row(r+1,c+1); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
    for idx_offset,(r,c) in enumerate(v_vars):
        idx=num_h+idx_offset
        if r>0:
            row=cell_row(r-1,c); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if r+2 < n:
            row=cell_row(r+2,c); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if c>0:
            row=cell_row(r,c-1)+cell_row(r+1,c-1); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
        if c+1 < n:
            row=cell_row(r,c+1)+cell_row(r+1,c+1); row[idx]-=1; rows.append(row); lbs.append(0); ubs.append(BIG)
    A=np.array(rows,float); lb=np.array(lbs,float); ub=np.array(ubs,float)
    c_vec=-np.ones(num_vars)
    integrality=np.ones(num_vars,dtype=int)
    bounds=Bounds(0,1)
    res=opt.milp(c=c_vec, constraints=LinearConstraint(A,lb,ub), bounds=bounds, integrality=integrality, options={"disp": False})
    if not res.success:
        return None
    return int(-res.fun)  # number of dominos
