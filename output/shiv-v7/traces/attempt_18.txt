# Blocking constraints
def add_block_constraint(placement_idx, needed_cells):
    # needed_cells: list of cell indices that must be occupied
    k = len(needed_cells)
    if k == 0:
        return  # automatically satisfied (edge)
    row = np.zeros(total_vars)
    # sum of x covering needed cells
    # For each needed cell, we need sum of placements covering that cell
    # We'll add coefficients for each placement that covers any needed cell
    # Since each cell can be covered by at most one placement, we can just add 1 for each placement covering that cell
    # However a placement may cover two needed cells (if both needed cells are part of same domino). That would add coefficient 2, which is okay because then if that placement is selected, it occupies both cells, counting as 2 occupied cells.
    for ci in needed_cells:
        for pj in cell_to_placements[ci]:
            row[pj] += 1
    # RHS: k * x_i
    row[placement_idx] = -k  # move to LHS: sum - k*x_i >= 0
    A_rows.append(row)
    lb.append(0)
    ub.append(np.inf)
