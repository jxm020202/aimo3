import itertools

def min_f_bruteforce(n, final):
    min_f=10**9
    best=None
    # generate nondecreasing sequences of length n with values 1..final
    # brute force using recursion limited small n
    def rec(idx, prev, seq):
        nonlocal min_f, best
        if idx==n:
            if seq[-1]==final:
                f=compute_f(seq)
                if f<min_f:
                    min_f=f
                    best=seq[:]
            return
        # remaining positions
        for val in range(prev, final+1):
            seq.append(val)
            rec(idx+1, val, seq)
            seq.pop()
    rec(1,1,[1]) # start with a1=1 already in seq
    return min_f, best

import itertools

def min_f_bruteforce(n, M):
    # n odd
    # generate monotone sequences with a1=1, an=M
    # we can generate all sequences of length n with values in [1,M]
    # but monotone reduces possibilities.
    # Use recursion to generate.
    best = None
    best_seq = None
    def rec(idx, prev, seq):
        nonlocal best, best_seq
        if idx == n:
            # sequence complete
            f = sum(x*x for x in seq) - sum(seq[i]*seq[i+2] for i in range(n-2))
            if best is None or f < best:
                best = f
                best_seq = seq[:]
            return
        # determine allowed range
        lo = prev
        hi = M if idx < n-1 else M  # final must be M
        if idx == n-1:
            lo = hi = M
        for val in range(lo, hi+1):
            seq.append(val)
            rec(idx+1, val, seq)
            seq.pop()
    rec(0, 1, [])
    return best, best_seq

    best = None
    best_seq = None
    # generate nondecreasing sequences length n with a1=1, a_n=max_val
    # Use recursion to generate

        nonlocal best, best_seq
        if pos == n:
            if seq[-1] != max_val:
                return
            # compute f
            a = seq
            f = sum(x*x for x in a) - sum(a[i]*a[i+2] for i in range(n-2))
            if best is None or f < best:
                best = f
                best_seq = a[:]
            return

        # pos is current index (0-based)
        # we need to fill a[pos]
        # ensure monotonic
        for val in range(prev, max_val+1):
            # prune if remaining positions cannot reach max_val
            # we need to ensure that we can still reach max_val at end
            # maximum possible increase per step is unlimited, so just ensure that we can set later values to max_val
            # always possible
            seq.append(val)
            rec(pos+1, val, seq)
            seq.pop()

def brute_opt(n, max_val):
    best = None
    best_seq = None
    # generate nondecreasing sequences length n with a1=1, a_n=max_val
    # Use recursion with pruning maybe for small n.
    def rec(idx, prev, seq, remaining_sum):
        nonlocal best, best_seq
        if idx == n:
            if seq[-1] != max_val:
                return
            f = sum(x*x for x in seq) - sum(seq[i]*seq[i+2] for i in range(n-2))
            if best is None or f < best:
                best = f
                best_seq = seq[:]
            return
        # remaining positions = n - idx
        # we need to be able to reach max_val at end; we can always set later to max_val.
        for val in range(prev, max_val+1):
            # simple prune: if val > max_val: break
            seq.append(val)
            rec(idx+1, val, seq, remaining_sum)
            seq.pop()
    rec(1, 1, [1], None)
    return best, best_seq

def min_f_dp(N, max_val):
    INF = 10**18
    dp = [[INF]*(max_val+1) for _ in range(max_val+1)]
    a1=1
    for a2 in range(a1, max_val+1):
        dp[a1][a2] = a1*a1 + a2*a2
    for i in range(2, N):
        newdp = [[INF]*(max_val+1) for _ in range(max_val+1)]
        for prevprev in range(1, max_val+1):
            row = dp[prevprev]
            for prev in range(prevprev, max_val+1):
                cur = row[prev]
                if cur == INF:
                    continue
                # next value
                if i+1 == N:
                    next_vals = [max_val]
                else:
                    next_vals = range(prev, max_val+1)
                for nxt in next_vals:
                    newcost = cur + nxt*nxt - prevprev*nxt
                    if newcost < newdp[prev][nxt]:
                        newdp[prev][nxt] = newcost
        dp = newdp
    ans = INF
    for a_prev in range(1, max_val+1):
        val = dp[a_prev][max_val]
        if val < ans:
            ans = val
    return ans

def construct_seq(n,M):
    # n odd, M final odd value
    p = (n+1)//2
    # number of odd gaps = p-1
    gaps = p-1
    total_inc = M-1
    # allocate increments: as many 1's as possible, remainder distributed as larger increments later
    incs = [1]*min(total_inc, gaps)
    remaining = total_inc - sum(incs)
    # distribute remaining across last gap(s) by adding to last inc(s)
    idx = len(incs)-1
    while remaining>0:
        incs[idx] += 1
        remaining -= 1
        idx -= 1
        if idx < 0:
            idx = len(incs)-1
    # compute odd values
    odd_vals = [1]
    for inc in incs:
        odd_vals.append(odd_vals[-1]+inc)
    # fill remaining odd positions with last value
    while len(odd_vals) < p:
        odd_vals.append(odd_vals[-1])
    # construct full sequence with evens copying previous odd
    seq = []
    for k in range(p):
        seq.append(odd_vals[k])  # odd
        if len(seq) < n:
            seq.append(odd_vals[k])  # even copy
    return seq

    n=len(seq)
    sum_sq=sum(x*x for x in seq)
    sum_cross=sum(seq[i]*seq[i+2] for i in range(n-2))
    return sum_sq - sum_cross

def min_f_bruteforce(n, max_val=5):
    # brute force all monotone sequences of length n with a1=1, an=max_val (or given)
    # limited to small max_val for feasibility
    from itertools import product
    best=10**9
    best_seq=None
    # generate nondecreasing sequences using recursion
    def rec(idx, prev, seq):
        nonlocal best,best_seq
        if idx==n:
            # seq length n, compute f
            f=compute_f(seq)
            if f<best:
                best=f
                best_seq=seq[:]
            return
        # remaining positions: need to reach final value max_val at last index
        # we can allow any value >=prev up to max_val
        for val in range(prev, max_val+1):
            seq.append(val)
            rec(idx+1, val, seq)
            seq.pop()
    # start with a1=1
    rec(1,1,[1])
    return best,best_seq

def min_f_dp(n=2025, total_inc=198):
    # d indices from 1 to n-1 (2024)
    # DP[i][prev_d][used] = min cost up to position i (i processed up to d_i), with prev_d = d_i, used sum so far = used
    # We'll use dict of dicts for each i
    from collections import defaultdict
    # Initialize for i=1 (d1)
    dp = {}
    for d1 in range(total_inc+1):
        used = d1
        if used > total_inc: continue
        cost = (1 + d1)**2  # initial term
        dp[(d1, used)] = cost
    # iterate i from 2 to n-1 (i.e., d_i for i=2..n-1)
    for i in range(2, n):  # i corresponds to d_i, up to n-1 inclusive? n=2025, so i up to 2024
        new_dp = {}
        max_d = total_inc  # d_i cannot exceed total_inc
        for (prev_d, used), cost in dp.items():
            remaining = total_inc - used
            # d_i can be from 0 to remaining
            for cur_d in range(remaining+1):
                new_used = used + cur_d
                new_cost = cost + (prev_d + cur_d)**2
                key = (cur_d, new_used)
                if key not in new_dp or new_cost < new_dp[key]:
                    new_dp[key] = new_cost
        dp = new_dp
        # optional: prune large dict size
        # print(i, len(dp))
    # after processing d_{n-1} (i=n-1 = 2024), add final term (199 - d_{n-1})^2
    best = None
    for (prev_d, used), cost in dp.items():
        if used != total_inc:
            continue
        total_cost = cost + (199 - prev_d)**2
        if best is None or total_cost < best:
            best = total_cost
    # final f = 0.5 * best + 0.5*(1+199**2) (fixed part)
    fixed = (1 + 199**2)  # 1 + 39601 = 39602
    f = 0.5 * best + 0.5 * fixed
    return f, best

def solve_continuous(m=2024, total_inc=198):
    # Build H matrix tridiagonal with 4 on diag, 4 on off-diag
    diag = np.full(m, 4.0)
    off = np.full(m-1, 4.0)
    H = np.diag(diag) + np.diag(off, k=1) + np.diag(off, k=-1)
    # vector u: -c where c_1=2, c_m=-398
    u = np.zeros(m)
    u[0] = -2.0
    u[-1] = 398.0
    # Solve H p = u
    p = np.linalg.solve(H, u)
    # Solve H q = 1
    ones = np.ones(m)
    q = np.linalg.solve(H, ones)
    lam = (p.sum() - total_inc) / q.sum()
    d = p - lam * q
    return d, lam, p, q

            from collections import deque
            hull = deque()

            def bad(l1, l2, l3):
                m1,b1,_ = l1

                    hull.pop()
                hull.append(line)

                continue
            def value(line, x):
                m,b,_ = line
                return m*x + b
            # query
            for nxt in range(curr, V+1):
                while len(hull) >= 2 and value(hull[0], nxt) >= value(hull[1], nxt):

                best_val = value(best_line, nxt)
                DP2[curr][nxt] = nxt*nxt + best_val
                # store predecessor: previous state had (prev, curr) where prev = best_line's prev
                _,_,prev_of_best = best_line
                ptr2[curr][nxt] = (prev_of_best, curr)
        DP = DP2
        prev_ptr = ptr2

        seq[i-2] = prev
    seq[0] = 1
    return best, seq

    return sum(x*x for x in seq) - sum(seq[i]*seq[i+2] for i in range(n-2))

    for _ in range(trials):

        f = compute_f(seq)

    return best

def min_f_bruteforce(n, start, end):
    # generate nondecreasing integer sequences length n with given endpoints
    # values between start and end inclusive
    # brute force via recursion (may be large)
    best = None
    best_seq=None
    def rec(pos, prev, seq):
        nonlocal best,best_seq
        if pos==n:
            if seq[-1]==end:
                f=compute_f(seq)
                if best is None or f<best:
                    best=f
                    best_seq=seq[:]
            return
        # remaining positions including current: n-pos
        # we need to ensure we can reach end with nondecreasing steps
        # max possible increase remaining = end - prev
        # we can set current value from prev to end
        for val in range(prev, end+1):
            seq.append(val)
            rec(pos+1, val, seq)
            seq.pop()
    rec(0, start, [])
    return best,best_seq
