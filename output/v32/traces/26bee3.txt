def check_solution(selected):
    # selected is list of edge dicts
    for a_idx, a in enumerate(selected):
        for b_idx in range(a_idx+1, len(selected)):
            b = selected[b_idx]
            # condition: either end of a == start of b or end of b == start of a
            if a['end'] == b['start'] or b['end'] == a['start']:
                continue
            # else distance between ends >=2
            dx = a['end'][0] - b['end'][0]
            dy = a['end'][1] - b['end'][1]
            dist_sq = dx*dx + dy*dy
            if dist_sq < 4:  # less than 2
                return False, (a,b,dist_sq)
    return True, None

def max_edges_subgrid(col_start, col_end):
    # columns inclusive range
    cols = list(range(col_start, col_end+1))
    # generate edges whose heads are in these columns (and rows 1..16)
    sub_edges = []
    for i in range(16):
        for j in range(16):
            # NE edge
            tail = (i,j)
            head = (i+1,j+1)
            if head[0] in cols and head[1]>=1:
                sub_edges.append(((tail, head)))
            # NW edge
            tail = (i+1,j)
            head = (i,j+1)
            if head[0] in cols and head[1]>=1:
                sub_edges.append(((tail, head)))
    # conflict pairs as before
    tails_sub = [e[0] for e in sub_edges]
    heads_sub = [e[1] for e in sub_edges]
    def conflict_sub(i,j):
        hx1, hy1 = heads_sub[i]
        hx2, hy2 = heads_sub[j]
        dx = hx1 - hx2
        dy = hy1 - hy2
        if dx*dx + dy*dy >= 4:
            return False
        if heads_sub[i] == tails_sub[j] or heads_sub[j] == tails_sub[i]:
            return False
        return True
    # Build MILP
    m = len(sub_edges)
    c = -np.ones(m)
    # constraints
    pairs = [(i,j) for i in range(m) for j in range(i+1,m) if conflict_sub(i,j)]
    A = np.zeros((len(pairs), m), dtype=int)
    b = np.ones(len(pairs), dtype=int)
    for idx, (i,j) in enumerate(pairs):
        A[idx,i]=1
        A[idx,j]=1
    bounds = Bounds(0,1)
    integrality = np.ones(m, dtype=int)
    res = milp(c=c, constraints=LinearConstraint(A, -np.inf, b), bounds=bounds,
               integrality=integrality, options={'disp': False})
    return -res.fun

def conflict(d1, d2):
    # d1,d2 are tuples ((i,j,dir), start, end)
    _, s1, e1 = d1

def max_edges_dp(rows=17, cols=16):
    # generate all patterns (bitmask) of length rows with no adjacent 1s
    patterns = []
    for mask in range(1<<rows):
        # check no adjacent 1s
        if mask & (mask<<1):
            continue
        patterns.append(mask)
    # precompute compatibility between patterns for adjacent columns: no 1s in same row or adjacent rows
    compat = {}
    for p in patterns:
        compat[p] = []
        for q in patterns:
            # check no rows within distance 1: i.e., (p & q) ==0 and (p<<1 & q)==0 and (p>>1 & q)==0
            if (p & q) or ((p<<1) & q) or ((p>>1) & q:
                continue
            compat[p].append(q)
    # DP
    dp = {p: bin(p).count('1') for p in patterns}  # first column
    for col in range(1, cols):
        new_dp = {}
        for p in patterns:
            best = -1
            for q in compat[p]:
                val = dp.get(q, -1)
                if val >=0:
                    cand = val + bin(p).count('1')
                    if cand > best:
                        best = cand
            if best >=0:
                new_dp[p] = best
        dp = new_dp
    return max(dp.values())

import itertools, math

def min_dist(k):
    min_d = float('inf')
    for moves in itertools.product([(1,1),(1,-1),(-1,1),(-1,-1)], repeat=k):
        dx = sum(m[0] for m in moves)
        dy = sum(m[1] for m in moves)
        d = math.hypot(dx, dy)
        if d < min_d:
            min_d = d
    return min_d
