# generate candidate complex numbers from roots of unity up to order 6, plus maybe some other simple numbers
def roots_of_unity(n):
    return [cmath.exp(2j*math.pi*k/n) for k in range(n)]

candidates = set()
candidates.add(0)
for n in range(1,7):
    for z in roots_of_unity(n):
        # exclude -1 (z == -1)
        if abs(z+1) < 1e-9:
            continue
        candidates.add(z)
# also include -0.5 maybe
candidates.add(-0.5)
candidates = list(candidates)

def satisfies(rs, m):
    for j in range(1, m+1):
        prod = 1
        for r in rs:
            prod *= (r**j + 1)
        if abs(prod - 1) > 1e-6:
            return False
    return True

def find_max_m(t_max=4):
    best = 0
    best_set = None
    for t in range(1, t_max+1):
        for rs in itertools.combinations_with_replacement(candidates, t):
            # ensure not all zero
            if all(abs(r) < 1e-9 for r in rs):
                continue
            # try increasing m
            m=1
            while True:
                if satisfies(rs, m):
                    m+=1
                else:
                    break
            # after loop, m fails, so max satisfied is m-1
            if m-1 > best:
                best = m-1
                best_set = rs
    return best, best_set

def construct_residue_set(p):
    # p prime, p odd
    zeta = cmath.exp(2j*math.pi/p)
    residues = [pow(a,2,p) for a in range(1,p) if a*a % p != 0]  # Actually squares mod p
    # get distinct non-zero squares
    squares = sorted(set([ (a*a) % p for a in range(1,p) ]))
    # exclude 0
    squares = [s for s in squares if s != 0]
    # map to powers
    return [zeta**s for s in squares]

    return True, None, None

We see for N=9, product = 4 for e=3,6 (multiples of 3). For N=15, product = 4 for e multiples of 3, product = 16 for e multiples of 5, etc. So product deviates from 1 when e shares a factor with N.

Thus to have product =1 for all e up to m, we need m less than the smallest prime divisor of N (since at e equal to that divisor, product will be >1). For prime N, smallest divisor is N itself, so product =1 for e=1..N-1.

Thus maximal m achievable with this construction is N-1 where N is an odd prime ≤ 646.

Thus maximal m = largest prime ≤ 646 minus 1 = 643 - 1 = 642.

Now can we achieve m larger than 642 using a different construction not based on full set of roots? Perhaps we could use a combination of roots of different orders to extend range beyond prime-1.

But we have limit of 645 variables. Could we use more than N-1 numbers to achieve longer range? For example, we could use all non-zero roots of unity for several primes and combine them, perhaps achieving product =1 for e up to LCM of primes minus 1? Let's think.

If we take union of sets of roots for different primes, the product for each e will be product of contributions from each set. For each prime p_i, the contribution is 1 for e not divisible by p_i, and 2^{p_i-1} for e divisible by p_i. So overall product will be 1 unless e is divisible by any of the primes, in which case product will be multiplied by 2^{p_i-1} for each prime dividing e (maybe multiple). So product will be 1 only for e that are not divisible by any of the chosen primes. So the set of e for which product=1 is the set of integers ≤ m that are coprime to all chosen primes, i.e., coprime to their product (the set of numbers with no prime factor among chosen primes). To have product=1 for all e up to m, we need that every integer 1..m is not divisible by any of the chosen primes, which is impossible if we include any prime ≤ m. So we must avoid primes ≤ m. That suggests we cannot extend beyond the largest prime minus 1.

But maybe we can use not all non-zero roots, but a subset that yields product 1 for more e.

Consider using all non-zero roots of unity for prime p, we get product 1 for e=1..p-1. Could we add extra numbers (maybe extra roots of unity of other orders) to cancel the factor 2^{p-1} at e=p, while preserving product 1 for earlier e? At e=p, the p-th roots all become 1, giving factor 2^{p-1}. If we add some numbers that at e=p produce factor 2^{-(p-1)} (i.e., product = 1/2^{p-1}) while at e< p they produce factor 1, then overall product becomes 1 for all e up to p.

Thus we could potentially extend m beyond p-1 by adding extra numbers that compensate at e=p.

Thus we need to find numbers r such that for e=1..p-1, product (1 + r^e) = 1 (i.e., they contribute neutral factor), but for e=p, product = 2^{-(p-1)}.

Is that possible? Let's think.

If we take a number r such that r^p = -1? Then (1 + r^p) = 0, not good.

If we take r = -1, then (1 + (-1)^e) = 0 for odd e, 2 for even e. Not good.

If we take r such that r^e = 0 for e=1..p-1? That would be r=0, giving factor 1 for all e, not helpful.

def product_subgroup_N(N, d):
    # find primitive root modulo N (if exists)
    # For composite N, may not have primitive root; use generator of multiplicative group if cyclic.
    # We'll just find an element of order phi(N) if group cyclic.
    phi = sp.totient(N)
    # find element of order phi
    for a in range(2, N):
        if math.gcd(a, N) != 1:
            continue
        # compute order
        order = 1
        val = a % N
        while val != 1:
            val = (val * a) % N
            order += 1
            if order > phi:
                break
        if order == phi:
            primitive = a
            break
    else:
        return None, None
    step = phi // d
    subgroup = [pow(primitive, step*k, N) for k in range(d)]
    omega = cmath.exp(2j*math.pi/N)
    prod = 1
    for h in subgroup:
        prod *= (1 + omega**h)
    return prod, subgroup
