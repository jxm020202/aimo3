        adj = [[0]*n for _ in range(n)]
        edge_count = 0

                adj[i][j] = adj[j][i] = 1
                edge_count += 1
        # check condition

                            break

                        found = True
                        break

def construct_graph(n=160, k=9):
    # core vertices 0..k-1
    # leaves assigned to groups
    t = n - 2*k
    q = t // k
    r = t % k
    # group sizes a_i = 1 + b_i, where b_i = q+1 for first r groups, q for rest
    a = []
    for i in range(k):
        b = q + (1 if i < r else 0)
        a_i = 1 + b
        a.append(a_i)
    # assign leaf vertices indices from k to n-1
    leaf_groups = []
    idx = k
    for size in a:
        group = list(range(idx, idx+size))
        leaf_groups.append(group)
        idx += size
    # adjacency matrix
    adj = [[0]*n for _ in range(n)]
    # core clique
    for i in range(k):
        for j in range(i+1, k):
            adj[i][j] = adj[j][i] = 1
    # core-leaf edges
    for i, group in enumerate(leaf_groups):
        for leaf in group:
            adj[i][leaf] = adj[leaf][i] = 1
    # leaf-leaf edges across groups (complete multipartite)
    for i in range(k):
        for j in range(i+1, k):
            for leaf_i in leaf_groups[i]:
                for leaf_j in leaf_groups[j]:
                    adj[leaf_i][leaf_j] = adj[leaf_j][leaf_i] = 1
    return adj

def check_condition(adj):

                        common = True

                if not common:
                    found = True

def compute_E(k, n=160):

    t = n - 2*k
    if t < 0:

    q = t // k
    r = t - k * q
    sum_b_sq = k * q * q + r * (2 * q + 1)
    sum_a_sq = k + 2 * t + sum_b_sq
    E = k*(k-1)//2 + m + (m*m - sum_a_sq)//2
    return E

def max_edges_n(n):
    # map edge (i,j) with i<j to index
    edge_idx = {}
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            edge_idx[(i,j)] = idx
            idx += 1
    m = idx
    # y variables for ordered pairs (u,v), u!=v
    y_idx = {}
    y_start = m
    for u in range(n):
        for v in range(n):
            if u==v: continue
            y_idx[(u,v)] = y_start
            y_start += 1
    total_vars = y_start
    # objective: maximize sum of edge variables
    c = np.zeros(total_vars)
    c[:m] = -1  # milp minimizes, so use -1 to maximize
    # constraints list
    A = []
    lb = []
    ub = []
    # 1. y_{u,v} <= edge_{u,v}
    for u in range(n):
        for v in range(n):
            if u==v: continue
            e = edge_idx[(min(u,v), max(u,v))]
            y = y_idx[(u,v)]
            row = np.zeros(total_vars)
            row[y] = 1
            row[e] = -1
            A.append(row)
            lb.append(-np.inf)
            ub.append(0)  # y - edge <= 0
    # 2. For each u, sum_v y_{u,v} >= 1
    for u in range(n):
        row = np.zeros(total_vars)
        for v in range(n):
            if u==v: continue
            row[y_idx[(u,v)]] = 1
        A.append(row)
        lb.append(1)
        ub.append(np.inf)
    # 3. triangle constraints: edge(u,w)+edge(v,w) <= 1 + (1 - y_{u,v})*M
    M = 2
    for u in range(n):
        for v in range(n):
            if u==v: continue
            for w in range(n):
                if w==u or w==v: continue
                # only need each unordered triple once? but okay.
                e_uw = edge_idx[(min(u,w), max(u,w))]
                e_vw = edge_idx[(min(v,w), max(v,w))]
                y_uv = y_idx[(u,v)]
                row = np.zeros(total_vars)
                row[e_uw] = 1
                row[e_vw] = 1
                row[y_uv] = M  # because we move (1 - y) * M to LHS: edge+edge - M*y <= 1 - M
                A.append(row)
                lb.append(-np.inf)
                ub.append(1 - M)  # RHS = 1 - M
    # variable bounds: edges binary 0/1, y binary 0/1
    bounds = Bounds(np.zeros(total_vars), np.ones(total_vars))
    # integrality: all variables integer
    integrality = np.ones(total_vars, dtype=int)
    # linear constraints
    A = np.array(A)
    lc = LinearConstraint(A, lb, ub)
    res = milp(c=c, constraints=lc, bounds=bounds, integrality=integrality, options={'disp': False})
    if not res.success:
        print("Failed", n, res.message)
        return None
    max_edges = -res.fun
    # round
    return int(round(max_edges))

def random_search(n, trials=100000):
    # start with complete bipartite floor/ceil
    a=n//2
    b=n-a
    # adjacency matrix
    adj=[[0]*n for _ in range(n)]
    for i in range(a):
        for j in range(a,n):
            adj[i][j]=adj[j][i]=1
    edges = [(i,j) for i in range(n) for j in range(i+1,n)]
    # compute current edge count
    cur = sum(adj[i][j] for i in range(n) for j in range(i+1,n))
    best=cur
    best_adj=None
    for t in range(trials):
        # randomly toggle an edge not present maybe
        i,j = random.choice(edges)
        # toggle
        adj[i][j] ^= 1
        adj[j][i] ^= 1
        if condition_ok(adj):
            cur = sum(adj[i][j] for i in range(n) for j in range(i+1,n))
            if cur>best:
                best=cur
                best_adj=[row[:] for row in adj]
        else:
            # revert toggle
            adj[i][j] ^= 1
            adj[j][i] ^= 1
    return best

def max_edges_milp(n):
    # create variable indices
    edge_index = {}
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            edge_index[(i,j)] = idx
            idx += 1
    m = idx
    # variable order: first x variables (edges), then w variables (private usage)
    # We'll create list of variable bounds and integrality.
    total_vars = m + m  # x and w
    # objective: maximize sum of x variables
    c = [0]*total_vars
    for i in range(m):
        c[i] = -1  # milp minimizes, so use -1 to maximize
    # bounds: all variables between 0 and 1
    bounds = [(0,1)]*total_vars
    # integrality: all variables integer (binary)
    integrality = [1]*total_vars
    
    # constraints: linear inequality A_ub x <= b_ub
    A = []
    b = []
    # constraint: w_{i,j} <= x_{i,j}
    for (i,j), e_idx in edge_index.items():
        row = [0]*total_vars
        row[e_idx] = -1  # -x_{i,j}
        row[m + e_idx] = 1   # + w_{i,j}
        A.append(row)
        b.append(0)  # w - x <= 0 => w <= x
    
    # constraint: for each vertex i, sum of w incident >= 1
    for i in range(n):
        row = [0]*total_vars
        for j in range(n):
            if i<j:
                e_idx = edge_index[(i,j)]
                row[m + e_idx] = 1
            elif j<i:
                e_idx = edge_index[(j,i)]
                row[m + e_idx] = 1
        # inequality: -sum w <= -1  (i.e., sum w >= 1)
        A.append([-coeff for coeff in row])
        b.append(-1)
    
    # constraint: for each private edge w_{i,j}=1, for each k != i,j: x_{i,k}+x_{j,k} <= 1
    # linearized as w_{i,j} + x_{i,k} + x_{j,k} <= 2
    for (i,j), e_idx in edge_index.items():
        w_idx = m + e_idx
        for k in range(n):
            if k==i or k==j:
                continue
            # x_{i,k}
            if i<k:
                x_ik = edge_index[(i,k)]
            else:
                x_ik = edge_index[(k,i)]
            # x_{j,k}
            if j<k:
                x_jk = edge_index[(j,k)]
            else:
                x_jk = edge_index[(k,j)]
            row = [0]*total_vars
            row[w_idx] = 1
            row[x_ik] = 1
            row[x_jk] = 1
            A.append(row)
            b.append(2)
    
    # Use scipy.optimize milp
    from scipy.optimize import milp, LinearConstraint, Bounds
    A = np.array(A, dtype=float)
    b = np.array(b, dtype=float)
    # Define constraints
    lin_con = LinearConstraint(A, -np.inf, b)
    bounds_obj = Bounds([0]*total_vars, [1]*total_vars)
    res = milp(c=c, constraints=lin_con, bounds=bounds_obj, integrality=integrality, options={'disp': False})
    if not res.success:
        print("MILP failed")
        return None
    x_vals = res.x[:m]
    # round to nearest integer
    edge_vals = [int(round(v)) for v in x_vals]
    total_edges = sum(edge_vals)
    return total_edges

    return best, best_adj

    m = len(pairs)
    # mapping for edge variable
    edge_idx = {}
    for idx,(i,j) in enumerate(pairs):
        edge_idx[(i,j)] = idx
        edge_idx[(j,i)] = idx  # both directions map to same index
    # ordered w variables: index = i*(n-1) + pos where pos is index of j in list of vertices != i
    # precompute mapping
    w_idx = {}
    idx_counter = m
    for i in range(n):
        pos = 0
        for j in range(n):
            if i==j: continue
            w_idx[(i,j)] = idx_counter
            idx_counter += 1
    total_vars = idx_counter
    import numpy as np
    c = np.zeros(total_vars)
    c[:m] = -1  # maximize edges
    # bounds binary
    bounds = Bounds(0, 1)
    # constraints
    A = []
    b = []
    # w_{i,j} <= x_{i,j}
    for i in range(n):

    # precompute mapping

    bounds = Bounds(0, 1)

def solve_milp(n):
    from scipy.optimize import milp, LinearConstraint, Bounds
    import numpy as np
    pairs = [(i,j) for i in range(n) for j in range(i+1,n)]

    for idx,(i,j) in enumerate(pairs):
        edge_idx[(i,j)] = idx
        edge_idx[(j,i)] = idx
    # w variables ordered
    w_idx = {}
    idx_counter = m
    for i in range(n):
        for j in range(n):
            if i==j: continue
            w_idx[(i,j)] = idx_counter
            idx_counter += 1
    total_vars = idx_counter
    c = np.zeros(total_vars)
    c[:m] = -1  # maximize edges
    bounds = Bounds(0,1)
    A = []
    b = []
    # w <= x
    for i in range(n):
        for j in range(n):
            if i==j: continue
            row = np.zeros(total_vars)
            row[w_idx[(i,j)]] = 1
            row[edge_idx[(i,j)]] = -1
            A.append(row); b.append(0)

    for i in range(n):

        for j in range(n):

        A.append(row); b.append(-1)
    # triangle condition
    for i in range(n):
        for j in range(n):
            if i==j: continue

def max_edges_n(n):
    # variables: x_{i,j} for i<j (binary)
    # y_{i,j} for i<j (binary) indicating triangle-free edge
    m = n*(n-1)//2
    # map edge index
    edge_index = {}
    idx = 0
    for i in range(n):
        for j in range(i+1,n):
            edge_index[(i,j)] = idx
            idx += 1
    # total variables = 2*m (x and y)
    total_vars = 2*m
    # objective: maximize sum x
    c = np.zeros(total_vars)
    for (i,j), e_idx in edge_index.items():
        c[e_idx] = -1  # milp minimizes, so negative for maximize
    # constraints list
    A = []
    b = []
    # y <= x
    for (i,j), e_idx in edge_index.items():
        row = np.zeros(total_vars)
        row[e_idx] = -1  # -x <= 0
        row[m + e_idx] = 1   # y - x <= 0  => y - x <= 0
        A.append(row)
        b.append(0)
    # triangle-free condition: for each edge (i,j) and each w != i,j:
    # x_{i,w}+x_{j,w} <= 2 - y_{i,j}
    for (i,j), e_idx in edge_index.items():
        for w in range(n):
            if w==i or w==j:
                continue
            # get indices for edges (i,w) and (j,w) with ordering
            a,b_ = (i,w) if i<w else (w,i)
            idx_iw = edge_index[(a,b_)]
            a,b_ = (j,w) if j<w else (w,j)
            idx_jw = edge_index[(a,b_)]
            row = np.zeros(total_vars)
            row[idx_iw] = 1
            row[idx_jw] = 1
            row[m + e_idx] = 1  # + y on LHS, move to RHS: x_iw + x_jw + y <= 2
            A.append(row)
            b.append(2)
    # each vertex must have at least one triangle-free incident edge: sum_{v} y_{u,v} >= 1
    for u in range(n):
        row = np.zeros(total_vars)
        for v in range(n):
            if u==v: continue
            i,j = (u,v) if u<v else (v,u)
            e_idx = edge_index[(i,j)]
            row[m + e_idx] = 1
        A.append(-row)  # -sum y <= -1  => sum y >=1
        b.append(-1)
    A = np.array(A)
    b = np.array(b)
    # bounds: binary 0-1
    bounds = Bounds(0, 1)
    # Use milp
    res = milp(c=c, constraints=LinearConstraint(A, -np.inf*np.ones_like(b), b), bounds=bounds, integrality=np.arange(total_vars))
    if not res.success:
        print("Failed", n, res.message)
        return None
    x = res.x[:m]
    # round to nearest integer
    edges = sum(round(val) for val in x)
    return edges

    edge_index = {}
    idx = 0
    for i in range(n):
        for j in range(i+1,n):

    rows = []

        row[m + e_idx] = 1
        row[e_idx] = -1
        rows.append(row); rhs.append(0)

def solve_milp_int(n):

    for i in range(n):
        for j in range(i+1,n):
            edge_index[(i,j)] = idx
            idx += 1
    m = len(edge_index)
    total_vars = 2*m
    c = np.zeros(total_vars)
    for (i,j), e_idx in edge_index.items():
        c[e_idx] = -1
    rows = []
    rhs = []
    # y <= x
    for (i,j), e_idx in edge_index.items():
        row = np.zeros(total_vars)
        row[m + e_idx] = 1
        row[e_idx] = -1
        rows.append(row); rhs.append(0)

def max_edges_n(n):
    max_e = 0
    best = []
    # enumerate all graphs via edge bitmask
    edges = [(i,j) for i in range(n) for j in range(i+1,n)]
    m = len(edges)
    for mask in range(1<<m):
        # quick prune: if potential max edges less than current max, skip? not needed for small n
        e_set = set()
        for idx, (i,j) in enumerate(edges):
            if mask>>idx & 1:
                e_set.add((i,j))
        # compute adjacency list
        adj = [set() for _ in range(n)]
        for (i,j) in e_set:
            adj[i].add(j)
            adj[j].add(i)
        # condition: for each vertex u, there exists neighbor v such that no common neighbor
        ok = True
        for u in range(n):
            found = False
            for v in adj[u]:
                # check common neighbor
                common = adj[u].intersection(adj[v])
                # exclude u and v themselves? they are not in adjacency sets
                if len(common)==0:
                    found = True
                    break
            if not found:
                ok = False
                break
        if ok:
            ecount = len(e_set)
            if ecount > max_e:
                max_e = ecount
                best = [mask]
            elif ecount == max_e:

def max_edges_n(n):
    max_e = 0
    best = []
    # enumerate all graphs via edge bitmask
    edges = [(i,j) for i in range(n) for j in range(i+1,n)]
    m = len(edges)
    for mask in range(1<<m):
        # quick prune: if potential max edges less than current max, skip? not needed for small n
        e_set = set()
        for idx, (i,j) in enumerate(edges):
            if mask>>idx & 1:
                e_set.add((i,j))
        # compute adjacency list
        adj = [set() for _ in range(n)]
        for (i,j) in e_set:
            adj[i].add(j)
            adj[j].add(i)
        # condition: for each vertex u, there exists neighbor v such that no common neighbor
        ok = True
        for u in range(n):
            found = False
            for v in adj[u]:
                # check common neighbor
                common = adj[u].intersection(adj[v])
                # exclude u and v themselves? they are not in adjacency sets
                if len(common)==0:
                    found = True
                    break
            if not found:
                ok = False
                break
        if ok:
            ecount = len(e_set)
            if ecount > max_e:
                max_e = ecount
                best = [mask]
            elif ecount == max_e:
                best.append(mask)
    return max_e, best

import itertools, math, numpy as np, scipy.optimize as opt, sys, time, random, math, collections, itertools, numpy as np
def max_edges_milp(n):
    # variable indices mapping
    edge_list = [(i,j) for i in range(n) for j in range(i+1,n)]
    m = len(edge_list)
    # map edge to index
    edge_index = {edge:i for i,edge in enumerate(edge_list)}
    # variables: e (binary) for edges, z (binary) for triangle-free edges
    num_vars = 2*m
    # objective: maximize sum e
    c = np.zeros(num_vars)
    c[:m] = -1  # minimize negative to maximize
    # constraints list
    A = []
    b = []
    # z <= e
    for i in range(m):
        row = np.zeros(num_vars)
        row[m+i] = 1   # z_i
        row[i] = -1    # -e_i
        A.append(row)
        b.append(0)
    # For each triple (u,v,w): z_uv <= 2 - e_uw - e_vw
    for u in range(n):
        for v in range(u+1,n):
            # index of uv
            idx_uv = edge_index[(u,v)]
            for w in range(n):
                if w==u or w==v:
                    continue
                # edges uw and vw
                a,bv = (min(u,w), max(u,w))
                idx_uw = edge_index[(a,bv)]
                a,bv = (min(v,w), max(v,w))
                idx_vw = edge_index[(a,bv)]
                row = np.zeros(num_vars)
                row[m+idx_uv] = 1   # z_uv
                row[idx_uw] = 1      # + e_uw
                row[idx_vw] = 1      # + e_vw
                # inequality: z_uv + e_uw + e_vw <= 2
                A.append(row)
                b.append(2)
    # For each vertex u: sum_{v} z_uv >= 1
    for u in range(n):
        row = np.zeros(num_vars)
        for v in range(n):
            if v==u: continue
            i,j = (min(u,v), max(u,v))
            idx = edge_index[(i,j)]
            row[m+idx] = 1
        A.append(-row)  # -sum z <= -1
        b.append(-1)
    # variable bounds: 0 <= var <= 1, integer
    bounds = [(0,1)]*num_vars
    integrality = np.ones(num_vars, dtype=int)
    # Use milp
    res = opt.milp(c, bounds=bounds, constraints=opt.LinearConstraint(np.array(A), -np.inf, np.array(b)), integrality=integrality, options={'presolve': True, 'disp': False})
    if not res.success:
        print("MILP failed")
        return None
    e_vals = np.round(res.x[:m])
    total_edges = int(e_vals.sum())
    return total_edges

def is_bipartite(adj):
    n=len(adj)
    color=[-1]*n
    from collections import deque
    for start in range(n):
        if color[start]==-1:
            color[start]=0
            q=deque([start])
            while q:
                u=q.popleft()
                for v in adj[u]:
                    if color[v]==-1:
                        color[v]=color[u]^1
                        q.append(v)
                    elif color[v]==color[u]:
                        return False
    return True

def max_edges_greedy(n, iterations=2000):
    best=0
    for _ in range(iterations):
        # start with empty graph but need each vertex to have at least one neighbor eventually.
        # start with random matching to ensure condition
        adj=[set() for _ in range(n)]
        vertices=list(range(n))
        random.shuffle(vertices)
        for i in range(0,n,2):
            if i+1<n:
                a,b=vertices[i],vertices[i+1]
                adj[a].add(b); adj[b].add(a)
        # random order edges
        edges=[(i,j) for i in range(n) for j in range(i+1,n) if j not in adj[i]]
        random.shuffle(edges)
        for (i,j) in edges:
            adj[i].add(j); adj[j].add(i)
            if not condition_ok_adj(adj):
                adj[i].remove(j); adj[j].remove(i)
        e=sum(len(s) for s in adj)//2
        if e>best:
            best=e
    return best
