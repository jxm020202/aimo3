def distinct_perm(n):
    for a in range(1,n):
        perm=[(a*i)%n for i in range(n)]
        diffs=[i-perm[i] for i in range(n)]
        sums=[i+perm[i] for i in range(n)]

def max_bipartite_matching(adj):
    # adj: list of lists from left to right indices
    nL = len(adj)
    nR = max((max(lst) if lst else -1) for lst in adj) + 1
    matchR = [-1]*nR
    def dfs(u, seen):
        for v in adj[u]:
            if seen[v]:
                continue
            seen[v]=True
            if matchR[v]==-1 or dfs(matchR[v], seen):
                matchR[v]=u
                return True
        return False
    result=0
    for u in range(nL):
        seen=[False]*nR

# compute a minimum vertex cover using König's algorithm (based on max matching)
def min_vertex_cover_bipartite(adj):
    # adj: left->list of right vertices
    nL = len(adj)
    nR = max((max(lst) if lst else -1) for lst in adj) + 1
    # compute maximum matching as before, also store matchL
    matchR = [-1]*nR
    matchL = [-1]*nL
    def dfs(u, seen):
        for v in adj[u]:
            if seen[v]:
                continue
            seen[v]=True
            if matchR[v]==-1 or dfs(matchR[v], seen):
                matchR[v]=u
                matchL[u]=v
                return True
        return False
    for u in range(nL):
        seen=[False]*nR
        dfs(u, seen)
    # Build alternating tree from unmatched left vertices
    # BFS/DFS from free left vertices using alternating edges (unmatched edges from left, matched edges from right)
    from collections import deque
    free_left = [u for u in range(nL) if matchL[u]==-1]
    visitedL = [False]*nL
    visitedR = [False]*nR
    q = deque(free_left)
    for u in free_left:
        visitedL[u]=True
    while q:
        u = q.popleft()
        # explore edges from left u that are not in matching
        for v in adj[u]:
            if not visitedR[v] and matchL[u]!=v:  # edge not matched
                visitedR[v]=True
                # if v is matched, go to its matched left vertex
                if matchR[v]!=-1 and not visitedL[matchR[v]]:
                    visitedL[matchR[v]]=True
                    q.append(matchR[v])
    # Minimum vertex cover = (L \ visitedL) ∪ (R ∩ visitedR)
    coverL = [u for u in range(nL) if not visitedL[u]]
    coverR = [v for v in range(nR) if visitedR[v]]
    return coverL, coverR, matchL, matchR

def bishop_domination(n):
    N=n*n
    # map (i,j) to index
    idx = lambda i,j: i*n + j
    # Build constraints matrix A (rows = squares, columns = squares)
    A = np.zeros((N,N), dtype=int)
    for i in range(n):
        for j in range(n):
            row = idx(i,j)
            # squares on same diagonals
            for p in range(n):
                for q in range(n):
                    if (p+q == i+j) or (p-q == i-j):
                        col = idx(p,q)
                        A[row, col] = 1
    # each row sum >= 1
    # objective c = 1 for each variable
    c = np.ones(N)
    # constraints: A x >= 1
    # Convert to LinearConstraint with lb=1, ub=inf
    # milp expects constraints in form A_ub x <= b_ub and A_eq x = b_eq, plus bounds.
    # We'll convert >= to <= by multiplying -1.
    A_ub = -A
    b_ub = -np.ones(N)
    # bounds 0 <= x <= 1
    bounds = Bounds(0, 1)
    # integrality: all variables integer (binary)
    integrality = np.ones(N, dtype=int)
    res = milp(c=c, constraints=LinearConstraint(A_ub, -np.inf, b_ub), bounds=bounds, integrality=integrality, options={'disp': False})
    if not res.success:
        print("Failed", n, res.message)
        return None
    x = np.rint(res.x).astype(int)
    return int(x.sum()), x.reshape((n,n))

def max_matching_size(n, color):
    # color 0 black, 1 white
    # Build bipartite graph: left vertices = even/odd i-j, right = even/odd i+j
    # edges = squares of given color
    left_vals = sorted(set(i-j for i in range(n) for j in range(n) if (i+j)%2==color))
    right_vals = sorted(set(i+j for i in range(n) for j in range(n) if (i+j)%2==color))
    left_index = {v:i for i,v in enumerate(left_vals)}
    right_index = {v:i for i,v in enumerate(right_vals)}
    L = len(left_vals)
    R = len(right_vals)
    # adjacency list
    adj = [[] for _ in range(L)]
    for i in range(n):
        for j in range(n):
            if (i+j)%2==color:
                adj[left_index[i-j]].append(right_index[i+j])
    # use simple max bipartite matching (Hungarian BFS)
    matchR = [-1]*R
    def dfs(u, seen):
        for v in adj[u]:
            if seen[v]:
                continue
            seen[v]=True
            if matchR[v]==-1 or dfs(matchR[v], seen):
                matchR[v]=u
                return True
        return False
    result=0
    for u in range(L):
        seen=[False]*R
        if dfs(u, seen):
            result+=1
    return result, L, R

def min_m_lower_bound(n):
    # lengths of sum-diagonals
    lengths = []
    for s in range(0, 2*n-1):
        length = min(s+1, 2*n-1 - s)
        lengths.append(length)
    total = sum(lengths) # = n^2
    # try each possible m, see if exists subset S of size <= m with sum of lengths of complement <= m
    for m in range(1, n+1):
        # we need to choose up to m diagonals to be covered directly
        # we want to minimize sum of lengths of remaining
        # best to choose the longest diagonals to be in S
        sorted_lengths = sorted(lengths, reverse=True)
        # choose top m lengths
        covered = sum(sorted_lengths[:m])
        remaining = total - covered
        if remaining <= m:
            return m
    return None
